Table Definitions

Book Table

CREATE TABLE book (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,
    author VARCHAR(255) NOT NULL,
    isbn VARCHAR(13) UNIQUE NOT NULL,
    available BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

Member Table

CREATE TABLE member (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    joined_date DATE NOT NULL,
    status ENUM('ACTIVE', 'SUSPENDED', 'EXPIRED') DEFAULT 'ACTIVE',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

Lending Table

CREATE TABLE lending (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    book_id BIGINT NOT NULL,
    member_id BIGINT NOT NULL,
    borrow_date DATE NOT NULL,
    due_date DATE NOT NULL,
    return_date DATE,
    status ENUM('BORROWED', 'RETURNED', 'OVERDUE') DEFAULT 'BORROWED',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (book_id) REFERENCES book(id),
    FOREIGN KEY (member_id) REFERENCES member(id)
);

Table Relationships

  • Book to Lending: One-to-Many
    • A book can be involved in multiple lending transactions
    • Each lending transaction involves exactly one book
  • Member to Lending: One-to-Many
    • A member can have multiple lending transactions
    • Each lending transaction involves exactly one member

Database Indexes

-- Book table indexes
CREATE INDEX idx_book_title ON book(title);
CREATE INDEX idx_book_author ON book(author);
CREATE UNIQUE INDEX idx_book_isbn ON book(isbn);

-- Member table indexes
CREATE UNIQUE INDEX idx_member_email ON member(email);
CREATE INDEX idx_member_status ON member(status);

-- Lending table indexes
CREATE INDEX idx_lending_book ON lending(book_id);
CREATE INDEX idx_lending_member ON lending(member_id);
CREATE INDEX idx_lending_status ON lending(status);
CREATE INDEX idx_lending_due_date ON lending(due_date);

Performance Optimization

Query Optimization Tips

  • Use appropriate indexes for frequently queried columns
  • Avoid SELECT * and only select needed columns
  • Use EXPLAIN to analyze query performance
  • Consider partitioning for large tables

Example EXPLAIN Output

EXPLAIN SELECT l.*, b.title, m.name 
FROM lending l
JOIN book b ON l.book_id = b.id
JOIN member m ON l.member_id = m.id
WHERE l.status = 'BORROWED';
